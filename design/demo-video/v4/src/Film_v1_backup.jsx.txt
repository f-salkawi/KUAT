const SPANS = Object.keys(START).map((k) => [START[k], lineEnd(k)])
function musicVolume(t) {
  let base = 0.42
  if (t >= 15.4) base = ramp(t, 15.4, 17.9, 0.42, 0.22, Easing.linear)
  if (t >= 17.9) base = ramp(t, 17.9, 18.4, 0.22, 0.03, Easing.linear)
  if (t >= IMPACT - 0.05) base = ramp(t, IMPACT - 0.05, IMPACT + 0.2, 0.03, 0.9, Easing.linear)
  if (t >= 21.5) base = ramp(t, 21.5, 24, 0.9, 0.62, Easing.linear)
  if (t >= SCENE.agentsEnd - 0.3 && t < MUSIC_BREAK) base = 0.85
  if (t >= MUSIC_BACK - 0.05) base = ramp(t, MUSIC_BACK + 0.6, MUSIC_BACK + 1.6, 0.85, 0.66, Easing.linear)
  const duck = Math.max(0, ...SPANS.map(([a, b]) => Math.min(ramp(t, a - 0.25, a + 0.05), ramp(t, b + 0.1, b + 0.6, 1, 0))))
  const depth = t > MUSIC_BACK - 0.1 && t < MUSIC_BACK + 0.8 ? 0.2 : 0.52
  return base * (1 - depth * duck) * ramp(t, END - 4.2, END - 0.3, 1, 0, Easing.linear)
}

function shake(t) {
  let x = 0, y = 0
  for (const at of [...QUOTES.map((q) => q.at), CLICKS[0]]) {
    if (t < at) continue
    const a = 7 * Math.exp(-(t - at) * 11)
    x += a * Math.sin((t - at) * 70 + at); y += a * Math.cos((t - at) * 63 + at * 2)
  }
  return [x, y]
}
